/*Para Arduino*/

Para que o Arduino funcione como uma câmera transmissora, você precisará de um módulo de câmera compatível e de um método para transmitir os dados capturados para outro dispositivo ou API. Aqui está um guia passo a passo para implementar isso:

1. Escolha do Módulo de Câmera
Use um módulo de câmera compatível com Arduino (OV7670, ArduCAM Mini)

2. Conexão do Módulo de Câmera
Conecte o módulo de câmera ao Arduino. Para o ArduCAM Mini, os pinos principais são:
(CS), (MOSI, MISO, SCK) (GND e VCC)


3. Configuração do Código
Use a biblioteca oficial do ArduCAM para capturar imagens e transmiti-las. Você pode instalar a biblioteca pelo Gerenciador de Bibliotecas do Arduino IDE.


/* Para capturar com ArduCAM e transmitir por SERIAL*/
#include <Wire.h>
#include <ArduCAM.h>
#include <SPI.h>

#define CS_PIN 10 // Pino CS do ArduCAM

ArduCAM myCAM(OV2640, CS_PIN);

void setup() {
    Serial.begin(115200); // Comunicação Serial
    Wire.begin();
    SPI.begin();

    // Inicializa a câmera
    myCAM.initCAM();
    myCAM.set_format(JPEG);
    myCAM.InitCAM();
    Serial.println("Câmera inicializada!");
}

void loop() {
    // Captura a imagem
    myCAM.flush_fifo();
    myCAM.start_capture();

    while (!myCAM.get_bit(ARDUCHIP_TRIG, CAP_DONE_MASK)) {
        delay(10); // Aguarda a captura
    }

    Serial.println("Imagem capturada!");

    // Envia os dados da imagem via Serial
    uint8_t temp;
    myCAM.CS_LOW();
    myCAM.set_fifo_burst();
    while (SPI.transfer(0x00) != 0xD9) { // 0xD9 é o marcador de fim de imagem JPEG
        temp = SPI.transfer(0x00);
        Serial.write(temp); // Envia byte a byte
    }
    myCAM.CS_HIGH();

    delay(5000); // Aguarda 5 segundos antes de capturar novamente
}
